option(ARTCRAFT_BUILD_NATIVE "Build the DomeD FLTK integration and example" ON)
set(ARTCRAFT_DOMED_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../dome-compositor" CACHE PATH
    "Canonical DomeD source providing the native integration")
if(ARTCRAFT_BUILD_NATIVE)
  foreach(source native/gpu-window.h native/fltk_appearance.h native/tray/desktop-menu.cpp)
    if(NOT EXISTS "${ARTCRAFT_DOMED_SOURCE}/${source}")
      message(FATAL_ERROR "Missing DomeD integration: ${ARTCRAFT_DOMED_SOURCE}/${source}")
    endif()
  endforeach()
  find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
  find_package(PkgConfig REQUIRED)
  find_package(OpenGL REQUIRED)
  pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET fontconfig freetype2 zlib gio-2.0)
  include("${ARTCRAFT_DOMED_SOURCE}/native/tray/desktop-menu.cmake")
  add_library(artcraft-platform src/application.cpp src/status-item.cpp)
  add_library(ArtCraft::Platform ALIAS artcraft-platform)
  target_include_directories(artcraft-platform
    PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
    PRIVATE ${ARTCRAFT_DOMED_SOURCE}/native ${ARTCRAFT_DOMED_SOURCE}/native/tray)
  target_link_libraries(artcraft-platform PUBLIC artcraft-commands fltk::gl-shared
    PRIVATE fltk::images-shared PkgConfig::NATIVE OpenGL::GL)
  target_compile_options(artcraft-platform PRIVATE -Wall -Wextra -Wpedantic)
  de_xr_desktop_menu(artcraft-platform)
  option(DOME_OFFICE_BUILD_LEGACY_FIXTURES "Build historical prototype editors and demo" OFF)
  include(GNUInstallDirs)
  if(DOME_OFFICE_BUILD_LEGACY_FIXTURES)
  add_executable(artcraft-base-demo examples/base-demo.cpp)
  target_link_libraries(artcraft-base-demo PRIVATE ArtCraft::Platform OpenGL::GL)
  set_target_properties(artcraft-base-demo PROPERTIES
    INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>"
    INSTALL_RPATH_USE_LINK_PATH TRUE)
  include(GNUInstallDirs)
  install(TARGETS artcraft-base-demo RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(PROGRAMS packaging/artcraft-demo DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES packaging/org.domed.ArtCraft.BaseDemo.desktop
    DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  endif()
  install(FILES LICENSE THIRD-PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/artcraft)
  install(FILES "${ARTCRAFT_DOMED_SOURCE}/LICENSE"
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/artcraft RENAME LICENSE.DomeD)
  install(FILES "${ARTCRAFT_DOMED_SOURCE}/native/bdf2x/UNLICENSE"
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/artcraft RENAME UNLICENSE.bdf2x)

  if(DOME_OFFICE_BUILD_LEGACY_FIXTURES)
  add_library(artcraft-editors STATIC apps/editor.cpp)
  target_link_libraries(artcraft-editors PUBLIC ArtCraft::Platform ArtCraft::Documents OpenGL::GL)
  target_compile_options(artcraft-editors PRIVATE -Wall -Wextra -Wpedantic)
  set(project_index 0)
  set(project_titles WordCraft GridCraft DeckCraft PhotoCraft VectorCraft)
  foreach(project wordcraft gridcraft deckcraft photocraft vectorcraft)
    list(GET project_titles ${project_index} APP_NAME)
    set(PROJECT_NAME_LOWER ${project})
    set(APP_ID org.domed.ArtCraft.${APP_NAME})
    configure_file(packaging/project-launcher.in launchers/${project} @ONLY)
    configure_file(packaging/project.desktop.in launchers/${APP_ID}.desktop @ONLY)
    add_executable(${project}-native apps/main.cpp)
    target_compile_definitions(${project}-native PRIVATE ARTCRAFT_PROJECT_INDEX=${project_index})
    target_link_libraries(${project}-native PRIVATE artcraft-editors)
    set_target_properties(${project}-native PROPERTIES
      INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>" INSTALL_RPATH_USE_LINK_PATH TRUE)
    math(EXPR project_index "${project_index} + 1")
    install(TARGETS ${project}-native RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT ${project})
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/launchers/${project} DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT ${project})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/launchers/${APP_ID}.desktop
      DESTINATION ${CMAKE_INSTALL_DATADIR}/applications COMPONENT ${project})
    install(FILES LICENSE THIRD-PARTY.md TEST-APPS.md
      DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/${project} COMPONENT ${project})
    install(FILES "${ARTCRAFT_DOMED_SOURCE}/LICENSE"
      DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/${project} RENAME LICENSE.DomeD COMPONENT ${project})
    install(FILES "${ARTCRAFT_DOMED_SOURCE}/native/bdf2x/UNLICENSE"
      DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/${project} RENAME UNLICENSE.bdf2x COMPONENT ${project})
  endforeach()

  endif()

    add_executable(photocraft-domed-native ports/photo-main.cpp)
    target_link_libraries(photocraft-domed-native PRIVATE dome-craft-port-host)
    target_compile_definitions(photocraft-domed-native PRIVATE DOME_PHOTO_REGISTRY="${CMAKE_CURRENT_SOURCE_DIR}/upstream.lock.json" DOME_PHOTO_ENGINE_DIR="${DOME_CRAFT_ENGINE_DIR}")
    target_compile_options(photocraft-domed-native PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
      add_executable(dome-photo-reliability-test tests/photo-reliability.cpp)

      target_compile_options(dome-photo-reliability-test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O0>)
      target_compile_definitions(dome-photo-reliability-test PRIVATE DOME_PHOTO_REGISTRY="${CMAKE_CURRENT_SOURCE_DIR}/upstream.lock.json" DOME_PHOTO_ENGINE_DIR="${DOME_CRAFT_ENGINE_DIR}")
